import numpy as np
import matplotlib.pyplot as plt

def get_rotation_matrix(theta_deg):
    theta = np.radians(theta_deg)
    return np.array([
        [np.cos(theta), -np.sin(theta), 0],
        [np.sin(theta), np.cos(theta), 0],
        [0, 0, 1]
    ])

def get_scaling_matrix(sx, sy, sz):
    return np.array([
        [sx, 0, 0],
        [0, sy, 0],
        [0, 0, sz]
    ])

def get_reflection_matrix():
    return np.array([
        [1, 0, 0],
        [0, 1, 0],
        [0, 0, -1]
    ])

# Original 3-D points
points = np.array([
    [1, 1, 1],
    [4, 1, 1],
    [2, 3, 1],
    [2, 2, 4],
    [1, 1, 1]
]).T

# Rotation, Scaling and Reflection
R = get_rotation_matrix(45)
S = get_scaling_matrix(2, 2, 2)
F = get_reflection_matrix()

T = F @ S @ R
new_points = T @ points

# Plot original and transformed points
fig = plt.figure()
ax = fig.add_subplot(111, projection='3d')

ax.plot(points[0], points[1], points[2],
        'b-o', label='Original')

ax.plot(new_points[0], new_points[1], new_points[2],
        'r-o', label='Transformed')

ax.set_title('3-D Geometric Transformation')
ax.set_xlabel('X-axis')
ax.set_ylabel('Y-axis')
ax.set_zlabel('Z-axis')
ax.legend()
plt.show()
