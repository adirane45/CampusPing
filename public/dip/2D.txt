import numpy as np
import matplotlib.pyplot as plt

def get_rotation_matrix(theta_deg):
    theta = np.radians(theta_deg)
    return np.array([
        [np.cos(theta), -np.sin(theta), 0],
        [np.sin(theta), np.cos(theta), 0],
        [0, 0, 1]
    ])

def get_scaling_matrix(sx, sy):
    return np.array([
        [sx, 0, 0],
        [0, sy, 0],
        [0, 0, 1]
    ])

def get_reflection_matrix(axis='x'):
    if axis == 'x':
        return np.array([
            [1, 0, 0],
            [0, -1, 0],
            [0, 0, 1]
        ])
    elif axis == 'y':
        return np.array([
            [-1, 0, 0],
            [0, 1, 0],
            [0, 0, 1]
        ])

# Original triangle
points = np.array([
    [1, 1, 1],
    [4, 1, 1],
    [2, 3, 1],
    [1, 1, 1]
]).T

# Rotation
R = get_rotation_matrix(45)
rotated = R @ points

# Scaling
S = get_scaling_matrix(2, 2)
scaled = S @ rotated

# Reflection
F = get_reflection_matrix('x')
reflected = F @ scaled

# Plot all transformations
plt.plot(points[0], points[1], 'b-o', label='Original')
plt.plot(rotated[0], rotated[1], 'g-o', label='Rotation')
plt.plot(scaled[0], scaled[1], 'm-o', label='Scaling')
plt.plot(reflected[0], reflected[1], 'r-o', label='Reflection')

plt.title('Composite 2-D Transformation')
plt.xlabel('X-axis')
plt.ylabel('Y-axis')
plt.grid()
plt.legend()
plt.axis('equal')
plt.show()
