Practical 5: Feature Selection using Correlation Heatmap

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from sklearn.datasets import load_breast_cancer


data = load_breast_cancer()

X = pd.DataFrame(data.data, columns=data.feature_names)


print("Shape of Dataset:")
print(X.shape)

print("\nFirst Five Records")
print(X.head())

print("\nFeature Names")
print(X.columns)

print("\nMissing Values")
print(X.isnull().sum())


correlation_matrix = X.corr()

print("\nCorrelation Matrix")

print(correlation_matrix)


plt.figure(figsize=(12,10))

plt.imshow(correlation_matrix,
           cmap='coolwarm',
           interpolation='nearest')

plt.colorbar()

plt.xticks(range(len(X.columns)),
           X.columns,
           rotation=90,
           fontsize=8)

plt.yticks(range(len(X.columns)),
           X.columns,
           fontsize=8)

plt.title("Correlation Heatmap")

plt.tight_layout()

plt.show()


threshold = 0.90

print("\nHighly Correlated Feature Pairs")

for i in range(len(correlation_matrix.columns)):
    for j in range(i):

        corr_value = correlation_matrix.iloc[i, j]

        if abs(corr_value) > threshold:

            print(f"{correlation_matrix.columns[i]}  <-->  "
                  f"{correlation_matrix.columns[j]} : "
                  f"{corr_value:.3f}")

upper = correlation_matrix.where(
    np.triu(np.ones(correlation_matrix.shape), k=1).astype(bool)
)

drop_columns = [column for column in upper.columns
                if any(abs(upper[column]) > threshold)]

print("\nFeatures to Remove")

print(drop_columns)


X_reduced = X.drop(columns=drop_columns)

print("\nOriginal Dataset Shape :", X.shape)

print("Reduced Dataset Shape  :", X_reduced.shape)

print("\nRemaining Features")

print(X_reduced.columns)
