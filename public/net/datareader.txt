using System;
using System.Data;
using System.Data.SqlClient;

namespace WebApplication16
{
    public partial class Default : System.Web.UI.Page
    {
        string cs = @"Data Source=(localdb)\MSSQLLocalDB;
                      Initial Catalog=CollegeDB;
                      Integrated Security=True";

        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                SimpleReader();
                ComplexReader();
                SimpleDataSet();
                ComplexDataSet();
            }
        }

        void SimpleReader()
        {
            SqlConnection con = new SqlConnection(cs);
            SqlCommand cmd = new SqlCommand(
                "SELECT StudentId, StudentName FROM Students", con);

            con.Open();
            SqlDataReader dr = cmd.ExecuteReader();
            gvReader.DataSource = dr;
            gvReader.DataBind();
            dr.Close();
            con.Close();
        }

        void ComplexReader()
        {
            SqlConnection con = new SqlConnection(cs);
            SqlCommand cmd = new SqlCommand(
                "SELECT Students.StudentName, Courses.CourseName " +
                "FROM Students INNER JOIN Courses " +
                "ON Students.CourseId = Courses.CourseId", con);

            con.Open();
            SqlDataReader dr = cmd.ExecuteReader();
            gvComplexReader.DataSource = dr;
            gvComplexReader.DataBind();
            dr.Close();
            con.Close();
        }

        void SimpleDataSet()
        {
            SqlConnection con = new SqlConnection(cs);
            SqlDataAdapter da = new SqlDataAdapter(
                "SELECT * FROM Students", con);

            DataSet ds = new DataSet();
            da.Fill(ds, "Students");

            gvDataSet.DataSource = ds.Tables["Students"];
            gvDataSet.DataBind();
        }

        void ComplexDataSet()
        {
            SqlConnection con = new SqlConnection(cs);
            SqlDataAdapter da = new SqlDataAdapter(
                "SELECT * FROM Students; SELECT * FROM Courses", con);

            DataSet ds = new DataSet();
            da.Fill(ds);

            ds.Tables[0].TableName = "Students";
            ds.Tables[1].TableName = "Courses";

            DataRelation relation = new DataRelation(
                "StudentCourse",
                ds.Tables["Courses"].Columns["CourseId"],
                ds.Tables["Students"].Columns["CourseId"]);

            ds.Relations.Add(relation);

            gvComplexDataSet.DataSource = ds.Tables["Students"];
            gvComplexDataSet.DataBind();
        }
    }
}
