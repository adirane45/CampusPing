using System;
using System.Data;

namespace WebApplication2
{
    public partial class Default : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                BindData();
            }
        }

        private DataTable GetStudents()
        {
            DataTable table = new DataTable();

            table.Columns.Add("ID");
            table.Columns.Add("Name");
            table.Columns.Add("Course");

            table.Rows.Add("1", "Rahul", "MCA");
            table.Rows.Add("2", "Aditya", "MCA");
            table.Rows.Add("3", "Priya", "MCA");
            table.Rows.Add("4", "Amit", "MCA");
            table.Rows.Add("5", "Sneha", "MCA");

            return table;
        }

        private void BindData()
        {
            DataTable table = GetStudents();

            dgStudents.DataSource = table;
            dgStudents.DataBind();

            dlStudents.DataSource = table;
            dlStudents.DataBind();
        }

        protected void dgStudents_EditCommand(
            object source,
            System.Web.UI.WebControls.DataGridCommandEventArgs e)
        {
            dgStudents.EditItemIndex = e.Item.ItemIndex;
            BindData();
        }

        protected void dgStudents_CancelCommand(
            object source,
            System.Web.UI.WebControls.DataGridCommandEventArgs e)
        {
            dgStudents.EditItemIndex = -1;
            BindData();
        }

        protected void dgStudents_UpdateCommand(
            object source,
            System.Web.UI.WebControls.DataGridCommandEventArgs e)
        {
            int index = e.Item.ItemIndex;

            DataTable table = GetStudents();

            table.Rows[index]["Name"] =
                ((System.Web.UI.WebControls.TextBox)e.Item.Cells[1].Controls[0]).Text;

            table.Rows[index]["Course"] =
                ((System.Web.UI.WebControls.TextBox)e.Item.Cells[2].Controls[0]).Text;

            dgStudents.EditItemIndex = -1;

            dgStudents.DataSource = table;
            dgStudents.DataBind();

            dlStudents.DataSource = table;
            dlStudents.DataBind();
        }
    }
}
