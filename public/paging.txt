using System;
using System.Data;


namespace DataGridDemo
{
    public partial class Default : System.Web.UI.Page
    {
        private DataTable GetStudents()
        {
            DataTable table = new DataTable();

            table.Columns.Add("ID", typeof(int));
            table.Columns.Add("Name");
            table.Columns.Add("Course");
            table.Columns.Add("City");

            table.Rows.Add(1, "Rahul", "MCA", "Jalgaon");
            table.Rows.Add(2, "Aditya", "MCA", "Pune");
            table.Rows.Add(3, "Priya", "MCA", "Mumbai");
            table.Rows.Add(4, "Amit", "MCA", "Nashik");
            table.Rows.Add(5, "Sneha", "MCA", "Jalgaon");
            table.Rows.Add(6, "Rohit", "MCA", "Pune");
            table.Rows.Add(7, "Neha", "MCA", "Mumbai");
            table.Rows.Add(8, "Karan", "MCA", "Nashik");

            return table;
        }

        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                BindGrid();
            }
        }

        private void BindGrid()
        {
            dgStudents.DataSource = GetStudents();
            dgStudents.DataBind();
        }

        protected void dgStudents_PageIndexChanged(
            object source,
            System.Web.UI.WebControls.DataGridPageChangedEventArgs e)
        {
            dgStudents.CurrentPageIndex = e.NewPageIndex;
            BindGrid();
        }

        protected void dgStudents_SortCommand(
            object source,
            System.Web.UI.WebControls.DataGridSortCommandEventArgs e)
        {
            DataTable table = GetStudents();

            table.DefaultView.Sort = e.SortExpression;

            dgStudents.DataSource = table;
            dgStudents.DataBind();
        }

        protected void btnFilter_Click(object sender, EventArgs e)
        {
            DataTable table = GetStudents();

            string search = txtSearch.Text.Trim();

            if (search != "")
            {
                DataView view = table.DefaultView;

                view.RowFilter =
                    "Name LIKE '%" + search.Replace("'", "''") + "%'";

                dgStudents.DataSource = view;
                dgStudents.DataBind();
            }
            else
            {
                BindGrid();
            }
        }

        protected void btnShowAll_Click(object sender, EventArgs e)
        {
            txtSearch.Text = "";
            dgStudents.CurrentPageIndex = 0;
            BindGrid();
        }
    }
}
