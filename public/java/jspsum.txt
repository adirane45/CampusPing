    <%
        int n = Integer.parseInt(request.getParameter("num"));

        n = Math.abs(n);
        int last = n % 10;
        int first = n;

        while (first >= 10) {
            first = first / 10;
        }

        int sum = first + last;
    %>

    <h2>Result</h2>

    <p style="color:red; font-size:18px;">
        Sum of first and last digit = <%= sum %>
