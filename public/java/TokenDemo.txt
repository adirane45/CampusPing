import java.util.StringTokenizer; // 1. Must import this

public class TokenDemo {
    public static void main(String[] args) {
        
        String sentence = "Advanced Java is very easy";
        
        // 2. Create the tokenizer (by default, it splits at spaces)
        StringTokenizer st = new StringTokenizer(sentence);
        
        System.out.println("Total words: " + st.countTokens());
        
        // 3. Loop through all the words and print them one by one
        while (st.hasMoreTokens()) {
            System.out.println(st.nextToken());
        }
    }
}
