using System;
using System.Collections;
using System.Data;

namespace DataBindingDemo
{
    public partial class Default : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                BindHashtable();
                BindArrayList();
                BindDataTable();
            }
        }

        // Hashtable
        private void BindHashtable()
        {
            Hashtable data = new Hashtable();

            data.Add("1", "Java");
            data.Add("2", "Python");
            data.Add("3", "C#");
            data.Add("4", "ASP.NET");

            ddlHashtable.DataSource = data;
            ddlHashtable.DataTextField = "Value";
            ddlHashtable.DataValueField = "Key";
            ddlHashtable.DataBind();
        }

        // ArrayList
        private void BindArrayList()
        {
            ArrayList data = new ArrayList();

            data.Add("Jalgaon");
            data.Add("Pune");
            data.Add("Mumbai");
            data.Add("Nashik");

            lstArrayList.DataSource = data;
            lstArrayList.DataBind();
        }

        // DataTable
        private void BindDataTable()
        {
            DataTable table = new DataTable();

            table.Columns.Add("ID");
            table.Columns.Add("Name");
            table.Columns.Add("Course");

            table.Rows.Add("1", "Rahul", "MCA");
            table.Rows.Add("2", "Aditya", "MCA");
            table.Rows.Add("3", "Priya", "MCA");

            gvDataTable.DataSource = table;
            gvDataTable.DataBind();
        }
    }
}
